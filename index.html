function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Timestamp", "Student Name", "Father/Roll No", "City", "Score", "Total", "Percentage", "Status", "Time Taken"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.timestamp || new Date().toLocaleString(),
    data.name || "",
    data.rollNo || "",
    data.city || "",
    data.score || 0,
    data.total || 100,
    data.percentage || "0%",
    data.status || "",
    data.timeTaken || ""
  ]);
  return ContentService.createTextOutput(JSON.stringify({ result: "success" })).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var results = [];
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    if (!r[1]) continue;
    results.push({
      submittedAt: String(r[0]),
      studentName: String(r[1]),
      rollNo: String(r[2]),
      city: String(r[3]),
      score: Number(r[4]) || 0,
      percentage: parseFloat(String(r[6]).replace('%', '')) || 0,
      passed: String(r[7]).toUpperCase() === "PASS",
      timeTakenFormatted: String(r[8]),
      sourceLabel: "Google Sheet"
    });
  }
  return ContentService.createTextOutput(JSON.stringify(results)).setMimeType(ContentService.MimeType.JSON);
}
